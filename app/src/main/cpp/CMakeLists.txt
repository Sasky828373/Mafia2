cmake_minimum_required(VERSION 3.22.1)
project(m2bootstrap LANGUAGES C CXX)

find_library(log-lib log)
find_library(dl-lib dl)
find_library(vulkan-lib vulkan)

add_library(m2bootstrap SHARED bootstrap.cpp recomp_runtime.cpp)
target_compile_features(m2bootstrap PRIVATE cxx_std_20)
target_compile_options(m2bootstrap PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(m2bootstrap PRIVATE ${log-lib} ${dl-lib} ${vulkan-lib})

# Separate ARM64 recomp core loaded by the Android host.
set(MAFIA2_RECOMP_SOURCES recomp_core.cpp recomp_runtime.cpp)
set(MAFIA2_HAS_GENERATED OFF)
set(MAFIA2_GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/mafia2_generated.cpp")
file(GLOB MAFIA2_GENERATED_SHARDS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/mafia2_generated_*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/generated/mafia2_generated_*.cpp")
if(MAFIA2_GENERATED_SHARDS)
    list(APPEND MAFIA2_RECOMP_SOURCES ${MAFIA2_GENERATED_SHARDS})
    set(MAFIA2_HAS_GENERATED ON)
    message(STATUS "Mafia II sharded generated dispatcher enabled")
elseif(EXISTS "${MAFIA2_GENERATED}")
    list(APPEND MAFIA2_RECOMP_SOURCES mafia2_generated.cpp)
    set(MAFIA2_HAS_GENERATED ON)
    message(STATUS "Mafia II generated dispatcher enabled")
else()
    message(STATUS "Mafia II generated dispatcher not present; building clean runtime shell")
endif()
add_library(mafia2_recomp SHARED ${MAFIA2_RECOMP_SOURCES})
if(MAFIA2_HAS_GENERATED)
    target_compile_definitions(mafia2_recomp PRIVATE M2_HAS_GENERATED_DISPATCH=1)
endif()
target_compile_features(mafia2_recomp PRIVATE cxx_std_20)
target_compile_options(mafia2_recomp PRIVATE -O3 -DNDEBUG -fvisibility=hidden)
target_link_libraries(mafia2_recomp PRIVATE ${log-lib} ${dl-lib})
